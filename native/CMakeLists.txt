cmake_minimum_required(VERSION 3.16)
# Ensure that Release Windows binaries use the static MSVC C runtime.
# This avoids requiring a separately installed VC++ Redistributable.
if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()
project(ys_sms_native C)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(ys_sms_bus STATIC sms_bus.c)
target_include_directories(ys_sms_bus PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_sms_vdp STATIC sms_vdp.c)
target_include_directories(ys_sms_vdp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_sms_clock STATIC sms_clock.c)
target_include_directories(ys_sms_clock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ys_sms_clock PUBLIC ys_sms_vdp)
add_library(ys_sms_video STATIC sms_video.c)
target_include_directories(ys_sms_video PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ys_sms_video PUBLIC ys_sms_vdp)
add_library(ys_video_fixture STATIC video_fixture.c)
target_include_directories(ys_video_fixture PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ys_video_fixture PUBLIC ys_sms_vdp)
add_library(ys_sms_sound STATIC sms_sound.c)
target_include_directories(ys_sms_sound PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_sms_input STATIC sms_input.c)
target_include_directories(ys_sms_input PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_z80 STATIC z80.c)
target_link_libraries(ys_z80 PUBLIC ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input)
add_library(ys_sms_runtime STATIC sms_runtime.c)
target_include_directories(ys_sms_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ys_sms_runtime PUBLIC ys_z80 ys_sms_clock)
if(MSVC)
    foreach(target ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input ys_sms_video ys_sms_clock ys_video_fixture ys_z80 ys_sms_runtime)
        target_compile_options(${target} PRIVATE /W4)
    endforeach()
else()
    foreach(target ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input ys_sms_video ys_sms_clock ys_video_fixture ys_z80 ys_sms_runtime)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endforeach()
endif()
set(YS_CHECKPOINT "M32" CACHE STRING "Source-backed native checkpoint")
set(YS_BUILD_CHANNEL "dev" CACHE STRING "Native build channel")
add_executable(ys-native-info build_info.c)
target_compile_definitions(ys-native-info PRIVATE
    "YS_CHECKPOINT=\"${YS_CHECKPOINT}\""
    "YS_BUILD_CHANNEL=\"${YS_BUILD_CHANNEL}\"")
add_executable(ys-native-preview video_preview.c)
target_link_libraries(ys-native-preview PRIVATE ys_sms_video ys_video_fixture)
# The primary Windows entry point is a persistent, self-contained GDI GUI.
# It displays synthetic hardware-test tiles (not a playable Ys game).
if(WIN32)
    add_executable(ys-native-window WIN32 window_win32.c)
    target_link_libraries(ys-native-window PRIVATE
        ys_sms_clock ys_sms_video ys_video_fixture user32 gdi32 shell32
    if(MSVC)
        target_compile_options(ys-native-window PRIVATE /W4)
    else()
        target_compile_options(ys-native-window PRIVATE
            -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
add_executable(ys-native-frames video_frames.c)
target_link_libraries(ys-native-frames PRIVATE
    ys_sms_clock ys_sms_video ys_video_fixture)
# SDL2 is optional: headless build and CI do not depend on system SDL2.
option(YS_ENABLE_SDL2_DISPLAY "Build optional SDL2 synthetic VDP window" ON)
if(YS_ENABLE_SDL2_DISPLAY)
    find_package(SDL2 QUIET)
    if(TARGET SDL2::SDL2)
        add_executable(ys-native-display display_sdl.c)
        target_link_libraries(ys-native-display PRIVATE
            ys_sms_clock ys_sms_video ys_video_fixture SDL2::SDL2)
    endif()
endif()
add_executable(ys-native-timing timing_demo.c)
target_link_libraries(ys-native-timing PRIVATE ys_sms_runtime)
add_executable(ys-native-timed-rom trace_timed.c)
target_link_libraries(ys-native-timed-rom PRIVATE ys_sms_runtime ys_sms_video)
add_executable(ys-native-trace trace_boot.c)
target_link_libraries(ys-native-trace PRIVATE ys_z80 ys_sms_video)
add_executable(ys-native-probe probe.c)
target_link_libraries(ys-native-probe PRIVATE ys_sms_bus)
enable_testing()
add_test(NAME build_info COMMAND ys-native-info --version)
if(WIN32)
    # Test the actual GUI-subsystem PE executable without a desktop session.
    add_test(NAME windows_gui_selftest COMMAND ys-native-window --self-test)
endif()
add_test(NAME video_preview COMMAND ys-native-preview --check)
add_test(NAME video_frames COMMAND ys-native-frames --check)
add_test(NAME timed_vblank COMMAND ys-native-timing --check)
add_executable(ys-sms-bus-tests test_bus.c)
target_link_libraries(ys-sms-bus-tests PRIVATE ys_sms_bus)
add_test(NAME sms_bus COMMAND ys-sms-bus-tests)
add_executable(ys-sms-vdp-tests test_vdp.c)
target_link_libraries(ys-sms-vdp-tests PRIVATE ys_sms_vdp)
add_test(NAME sms_vdp COMMAND ys-sms-vdp-tests)
add_executable(ys-sms-clock-tests test_clock.c)
target_link_libraries(ys-sms-clock-tests PRIVATE ys_sms_runtime)
add_test(NAME sms_clock COMMAND ys-sms-clock-tests)
add_executable(ys-z80-timing-tests test_timing.c)
target_link_libraries(ys-z80-timing-tests PRIVATE ys_sms_runtime)
add_test(NAME z80_timing COMMAND ys-z80-timing-tests)
add_executable(ys-sms-video-tests test_video.c)
target_link_libraries(ys-sms-video-tests PRIVATE ys_sms_video)
add_test(NAME sms_video COMMAND ys-sms-video-tests)
add_executable(ys-sms-sound-tests test_sound.c)
target_link_libraries(ys-sms-sound-tests PRIVATE ys_sms_sound)
add_test(NAME sms_sound COMMAND ys-sms-sound-tests)
add_executable(ys-sms-input-tests test_input.c)
target_link_libraries(ys-sms-input-tests PRIVATE ys_sms_input)
add_test(NAME sms_input COMMAND ys-sms-input-tests)
add_executable(ys-z80-tests test_z80.c)
target_link_libraries(ys-z80-tests PRIVATE ys_z80)
add_test(NAME z80_core COMMAND ys-z80-tests)
add_executable(ys-z80-irq-tests test_irq.c)
target_link_libraries(ys-z80-irq-tests PRIVATE ys_z80)
add_test(NAME z80_irq COMMAND ys-z80-irq-tests)

# CTest fixtures intentionally use assert() for both checks and setup.
# Force assert() on even in optimized Release / RelWithDebInfo packaging
# builds; otherwise -DNDEBUG compiles away setup and can crash tests.
foreach(target ys-sms-bus-tests ys-sms-vdp-tests ys-sms-clock-tests ys-sms-video-tests ys-sms-sound-tests
               ys-sms-input-tests ys-z80-tests ys-z80-irq-tests ys-z80-timing-tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
