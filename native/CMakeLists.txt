cmake_minimum_required(VERSION 3.16)
# Ensure that Release Windows binaries use the static MSVC C runtime.
# This avoids requiring a separately installed VC++ Redistributable.
if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()
project(ys_sms_native C)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

    foreach(target ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input ys_sms_video ys_sms_clock ys_video_fixture ys_z80 ys_sms_runtime ys_pc_game ys_pc_tiles ys_pc_records ys_pc_object_types ys_pc_objects ys_pc_scenes ys_pc_nametable)
        target_compile_options(${target} PRIVATE /W4)
    endforeach()

    foreach(target ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input ys_sms_video ys_sms_clock ys_video_fixture ys_z80 ys_sms_runtime ys_pc_game ys_pc_tiles ys_pc_records ys_pc_object_types ys_pc_objects ys_pc_scenes ys_pc_nametable)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endforeach()

option(YS_ENABLE_PC_SDL2 "Build enhanced native PC gameplay window" ON)
if(YS_ENABLE_SDL2_DISPLAY OR YS_ENABLE_PC_SDL2)
    find_package(SDL2 QUIET)
    if(TARGET SDL2::SDL2)
        if(YS_ENABLE_SDL2_DISPLAY)
            add_executable(ys-native-display display_sdl.c)
            target_link_libraries(ys-native-display PRIVATE
                ys_sms_clock ys_sms_video ys_video_fixture SDL2::SDL2)
        endif()
        if(YS_ENABLE_PC_SDL2)
            add_executable(ys-native-game game_sdl.c)
            target_link_libraries(ys-native-game PRIVATE ys_pc_game ys_pc_tiles ys_pc_records ys_pc_objects ys_pc_scenes ys_pc_nametable SDL2::SDL2)
            if(MSVC)
                target_compile_options(ys-native-game PRIVATE /W4)
            else()
                target_compile_options(ys-native-game PRIVATE
                    -Wall -Wextra -Wpedantic -Werror)
            endif()
        endif()
    endif()
endif()

# CTest fixtures intentionally use assert() for both checks and setup.
# Force assert() on even in optimized Release / RelWithDebInfo packaging
# builds; otherwise -DNDEBUG compiles away setup and can crash tests.
foreach(target ys-sms-bus-tests ys-sms-vdp-tests ys-sms-clock-tests ys-sms-video-tests ys-sms-sound-tests
               ys-sms-input-tests ys-z80-tests ys-z80-irq-tests ys-z80-timing-tests ys-pc-game-tests ys-pc-tiles-tests ys-pc-records-tests ys-pc-objects-tests ys-pc-object-types-tests ys-pc-scenes-tests ys-pc-nametable-tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
