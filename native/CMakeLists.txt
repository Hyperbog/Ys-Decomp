cmake_minimum_required(VERSION 3.16)
project(ys_sms_native C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(ys_sms_bus STATIC sms_bus.c)
target_include_directories(ys_sms_bus PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_sms_vdp STATIC sms_vdp.c)
target_include_directories(ys_sms_vdp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_sms_video STATIC sms_video.c)
target_include_directories(ys_sms_video PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ys_sms_video PUBLIC ys_sms_vdp)
add_library(ys_sms_sound STATIC sms_sound.c)
target_include_directories(ys_sms_sound PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_sms_input STATIC sms_input.c)
target_include_directories(ys_sms_input PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(ys_z80 STATIC z80.c)
target_link_libraries(ys_z80 PUBLIC ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input)
if(MSVC)
    foreach(target ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input ys_sms_video ys_z80)
        target_compile_options(${target} PRIVATE /W4)
    endforeach()
else()
    foreach(target ys_sms_bus ys_sms_vdp ys_sms_sound ys_sms_input ys_sms_video ys_z80)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endforeach()
endif()
set(YS_CHECKPOINT "M26" CACHE STRING "Source-backed native checkpoint")
set(YS_BUILD_CHANNEL "dev" CACHE STRING "Native build channel")
add_executable(ys-native-info build_info.c)
target_compile_definitions(ys-native-info PRIVATE
    "YS_CHECKPOINT=\"${YS_CHECKPOINT}\""
    "YS_BUILD_CHANNEL=\"${YS_BUILD_CHANNEL}\"")
add_executable(ys-native-preview video_preview.c)
target_link_libraries(ys-native-preview PRIVATE ys_sms_video)
add_executable(ys-native-trace trace_boot.c)
target_link_libraries(ys-native-trace PRIVATE ys_z80 ys_sms_video)
add_executable(ys-native-probe probe.c)
target_link_libraries(ys-native-probe PRIVATE ys_sms_bus)
enable_testing()
add_test(NAME build_info COMMAND ys-native-info --version)
add_test(NAME video_preview COMMAND ys-native-preview --check)
add_executable(ys-sms-bus-tests test_bus.c)
target_link_libraries(ys-sms-bus-tests PRIVATE ys_sms_bus)
add_test(NAME sms_bus COMMAND ys-sms-bus-tests)
add_executable(ys-sms-vdp-tests test_vdp.c)
target_link_libraries(ys-sms-vdp-tests PRIVATE ys_sms_vdp)
add_test(NAME sms_vdp COMMAND ys-sms-vdp-tests)
add_executable(ys-sms-video-tests test_video.c)
target_link_libraries(ys-sms-video-tests PRIVATE ys_sms_video)
add_test(NAME sms_video COMMAND ys-sms-video-tests)
add_executable(ys-sms-sound-tests test_sound.c)
target_link_libraries(ys-sms-sound-tests PRIVATE ys_sms_sound)
add_test(NAME sms_sound COMMAND ys-sms-sound-tests)
add_executable(ys-sms-input-tests test_input.c)
target_link_libraries(ys-sms-input-tests PRIVATE ys_sms_input)
add_test(NAME sms_input COMMAND ys-sms-input-tests)
add_executable(ys-z80-tests test_z80.c)
target_link_libraries(ys-z80-tests PRIVATE ys_z80)
add_test(NAME z80_core COMMAND ys-z80-tests)
add_executable(ys-z80-irq-tests test_irq.c)
target_link_libraries(ys-z80-irq-tests PRIVATE ys_z80)
add_test(NAME z80_irq COMMAND ys-z80-irq-tests)

# CTest fixtures intentionally use assert() for both checks and setup.
# Force assert() on even in optimized Release / RelWithDebInfo packaging
# builds; otherwise -DNDEBUG compiles away setup and can crash tests.
foreach(target ys-sms-bus-tests ys-sms-vdp-tests ys-sms-video-tests ys-sms-sound-tests
               ys-sms-input-tests ys-z80-tests ys-z80-irq-tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
