cmake_minimum_required(VERSION 3.16)
project(ys_sms_native C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(ys_sms_bus STATIC sms_bus.c)
target_include_directories(ys_sms_bus PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(ys_sms_bus PRIVATE /W4)
else()
    target_compile_options(ys_sms_bus PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ys-native-probe probe.c)
target_link_libraries(ys-native-probe PRIVATE ys_sms_bus)
enable_testing()
add_executable(ys-sms-bus-tests test_bus.c)
target_link_libraries(ys-sms-bus-tests PRIVATE ys_sms_bus)
add_test(NAME sms_bus COMMAND ys-sms-bus-tests)
