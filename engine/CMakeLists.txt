cmake_minimum_required(VERSION 3.16)
project(ys_expanded_engine C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(ys_content_registry STATIC content_registry.c)
target_include_directories(ys_content_registry PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(ys_content_registry PRIVATE /W4 /WX)
else()
    target_compile_options(ys_content_registry PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(ys-content-registry-tests tests/test_content_registry.c)
target_link_libraries(ys-content-registry-tests PRIVATE ys_content_registry)
if(MSVC)
    target_compile_options(ys-content-registry-tests PRIVATE /UNDEBUG /W4 /WX)
else()
    target_compile_options(ys-content-registry-tests PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME content_registry COMMAND ys-content-registry-tests)
